 if(state.query)p.set('q',state.query);
 if(state.year)p.set('year',state.year);
 if(state.tag)p.set('tag',state.tag);
 if(state.sort!=='source')p.set('sort',state.sort);
 if(state.practice)p.set('practice','1');
 try{history.replaceState(null,'',location.pathname+location.search+(p.size?'#'+p.toString():''));}catch{}
}
function syncControls(){
 $('search').value=state.query;$('year').value=state.year;$('tag').value=state.tag;$('sort').value=state.sort;$('practice').checked=state.practice;
 for(const button of $('nav').querySelectorAll('[data-category]'))button.setAttribute('aria-pressed',String(button.dataset.category===state.category));
 $('clear-search').hidden=!state.query;
 document.body.classList.toggle('sentence-view',state.category==='sentence');
}
function matches(record){
 if(state.category!=='all'&&record.category!==state.category)return false;
 if(state.year==='unknown'&&record.years.length)return false;
 if(state.year&&state.year!=='unknown'&&!record.years.includes(state.year))return false;
 if(state.tag&&!record.tags.includes(state.tag))return false;
 return normal(state.query).split(' ').filter(Boolean).every(token=>/^20\d{2}$/.test(token)?record.years.includes(token):record.search.includes(token));
}
function card(record){
 const article=make('article','entry');
 const top=make('div','entry-top');top.append(make('span','entry-category',names[record.category]));
 const uncertain=record.tags.find(t=>['待核验','暂定译法','语境判断','原稿疑点'].includes(t));
 if(uncertain)top.append(make('span','uncertain',uncertain));article.append(top);
 const title=make('h3','entry-title');title.append(make('span','term',record.term),make('span','colon','：'),make('span','meaning',state.practice?'展开下方，检查你的释义':record.meaning));article.append(title);
 const meta=make('div','meta');
 for(const y of record.years.length?record.years:['年份未标'])meta.append(make('span','year-chip'+(y===state.year?' active':''),y));
 for(const tag of record.tags)meta.append(make('span','tag',tag));article.append(meta);
 const details=make('details');details.append(make('summary','',state.practice?'显示释义与解析':'展开用法与出处'));
 const body=make('div','body');
 if(state.practice)body.append(make('p','answer',record.term+'：'+record.meaning));
 for(const node of record.nodes)body.append(safeCopy(node));
 const repo=document.documentElement.dataset.repository||'';
 if(/^https:\/\/github\.com\//.test(repo)){
  const link=make('a','source-link','查看这份 Markdown 原稿 ↗');link.href=repo.replace(/\/$/,'')+'/blob/HEAD/'+record.file.split('/').map(encodeURIComponent).join('/');link.target='_blank';link.rel='noopener noreferrer';body.append(link);
 }
 details.append(body);article.append(details);return article;
}
function render(){
 syncControls();writeHash();
 const found=records.filter(matches);
 if(state.sort==='alpha')found.sort((a,b)=>a.term.localeCompare(b.term,'en',{sensitivity:'base'}));
 else if(state.sort==='newest')found.sort((a,b)=>(Number(b.years.at(-1))||0)-(Number(a.years.at(-1))||0)||a.order-b.order);
 const pages=Math.max(1,Math.ceil(found.length/perPage));state.page=Math.min(Math.max(1,state.page),pages);
 const begin=(state.page-1)*perPage;const fragment=document.createDocumentFragment();
 for(const record of found.slice(begin,begin+perPage))fragment.append(card(record));
 $('entries').replaceChildren(fragment);$('empty').hidden=found.length>0;
 $('category-title').textContent=names[state.category];
 $('result-info').textContent=found.length?`找到 ${found.length} 条 · 当前 ${begin+1}—${Math.min(begin+perPage,found.length)} 条`:'找到 0 条';
 if(state.practice&&found.length)$('result-info').textContent+=' · 中文释义已遮住';
 $('page-info').textContent=state.page+' / '+pages;$('prev').disabled=state.page===1;$('next').disabled=state.page===pages;
 $('pagination').hidden=found.length<=perPage;
}

readHash();render();
